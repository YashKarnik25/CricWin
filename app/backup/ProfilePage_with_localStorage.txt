import React, { useContext, useEffect, useState, useRef } from "react";
import {
  View,
  Text,
  StyleSheet,
  ActivityIndicator,
  ScrollView,
  Animated,
  Dimensions,
  Image,
  TouchableOpacity,
  Modal,
} from "react-native";
import { ref, query, orderByChild, equalTo, get } from "firebase/database";
import { database } from "./firebase";
import { UserContext } from "./_layout";
import { useIsFocused } from "@react-navigation/native";
import { MaterialIcons } from '@expo/vector-icons';
import * as ImagePicker from 'expo-image-picker';  // Importing ImagePicker

const { height } = Dimensions.get("window");

const Profile = () => {
  const { emailGlobal } = useContext(UserContext);
  const [profileData, setProfileData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [selectedSticker, setSelectedSticker] = useState(require('../assets/images/profile/p1.jpg'));
  const [isModalVisible, setIsModalVisible] = useState(false);

  const matchesAnim = useRef(new Animated.Value(300)).current;
  const battingAnim = useRef(new Animated.Value(300)).current;
  const bowlingAnim = useRef(new Animated.Value(300)).current;

  const isFocused = useIsFocused();

  useEffect(() => {
    const fetchPlayerStats = async () => {
      try {
        const playerStatsRef = ref(database, "playerStats");
        const playerQuery = query(
          playerStatsRef,
          orderByChild("email"),
          equalTo(emailGlobal)
        );
        const snapshot = await get(playerQuery);
        if (snapshot.exists()) {
          const data = snapshot.val();
          const data2 = Object.values(data)[0];
          setProfileData(data2);
        } else {
          console.log("No record found for email", emailGlobal);
          setProfileData(null);
        }
      } catch (error) {
        console.error("Error fetching player stats:", error);
      }
      setLoading(false);
    };

    const fetchUserInfo = async () => {
      try {
        const userRef = ref(database, "user");
        const userQuery = query(
          userRef,
          orderByChild("email"),
          equalTo(emailGlobal)
        );
        const snapshot = await get(userQuery);
        if (snapshot.exists()) {
          const userData = snapshot.val();
          const userInfo = Object.values(userData)[0];
          setProfileData((prevData) => ({
            ...prevData,
            name: `${userInfo.firstname} ${userInfo.lastname}`,
            email: userInfo.email,
            contact: userInfo.contact,
            role: userInfo.role,
          }));
        } else {
          console.log("No user record found for email", emailGlobal);
        }
      } catch (error) {
        console.error("Error fetching user info:", error);
      }
    };

    fetchPlayerStats();
    fetchUserInfo();
  }, [emailGlobal]);

  useEffect(() => {
    const requestPermissions = async () => {
      const mediaPermission = await ImagePicker.requestMediaLibraryPermissionsAsync();
      const cameraPermission = await ImagePicker.requestCameraPermissionsAsync();
      if (mediaPermission.status !== 'granted' || cameraPermission.status !== 'granted') {
        alert('Permission to access media library or camera is required!');
      }
    };

    requestPermissions();
  }, []);

  useEffect(() => {
    if (isFocused) {
      matchesAnim.setValue(300);
      battingAnim.setValue(300);
      bowlingAnim.setValue(300);
      Animated.sequence([  
        Animated.timing(matchesAnim, {
          toValue: 0,
          duration: 500,
          useNativeDriver: true,
        }),
        Animated.timing(battingAnim, {
          toValue: 0,
          duration: 500,
          useNativeDriver: true,
        }),
        Animated.timing(bowlingAnim, {
          toValue: 0,
          duration: 500,
          useNativeDriver: true,
        }),
      ]).start();
    }
  }, [isFocused, matchesAnim, battingAnim, bowlingAnim]);

  const imageDictionary = [
    require('../assets/images/profile/p1.jpg'),
    require('../assets/images/profile/p2.jpg'),
    require('../assets/images/profile/p3.jpg'),
    require('../assets/images/profile/p4.jpg'),
  ];

  const handleStickerSelect = (image) => {
    setSelectedSticker(image);
    setIsModalVisible(false); 
  };

  const pickImage = async () => {
    
    try {
      const result = await ImagePicker.launchImageLibraryAsync({
        mediaTypes: ImagePicker.MediaTypeOptions.Images,
        allowsEditing: true,
        quality: 1,
      });

      if (!result.cancelled) {
        setSelectedSticker({ uri: result.uri }); 
        setIsModalVisible(false);
      }
    } catch (error) {
      console.error("Error picking image:", error);
    }
  };

  if (loading) {
    return (
      <View style={styles.loadingContainer}>
        <ActivityIndicator size="large" color="#4CAF50" />
      </View>
    );
  }

  if (!profileData) {
    return (
      <View style={styles.container}>
        <Text style={styles.title}>Profile</Text>
        <Text style={styles.noDataText}>
          No profile data found for {emailGlobal}
        </Text>
      </View>
    );
  }

  const { name, email, contact, role, batting, bowling, matches } = profileData;
  const battingData = batting || { runs: 'N/A', balls: 'N/A', strikeRate: 'N/A', average: 'N/A' };
  const bowlingData = bowling || { runs: 'N/A', balls: 'N/A', wickets: 'N/A' };

  return (
    <Animated.View style={{ flex: 1 }}>
      <ScrollView style={styles.container}>
        <Text style={styles.title}>Profile</Text>

        {/* Profile Picture Section */}
        <View style={styles.profileContainer}>
          <Image
            source={selectedSticker}
            style={styles.profilePicture}
          />
          <TouchableOpacity onPress={() => setIsModalVisible(true)} style={styles.editButton}>
            <MaterialIcons name="edit" size={30} color="white" />
          </TouchableOpacity>
        </View>

        {/* Player Details Section */}
        <View style={styles.tbl}>
          <Text style={styles.sectionTitle}>Played Information</Text>
          <View style={styles.tableRow}>
            <Text style={styles.detailText}>Name:</Text>
            <Text style={styles.fetchTextFet}>{name}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.detailText}>Email:</Text>
            <Text style={styles.fetchTextFet}>{email}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.detailText}>Contact:</Text>
            <Text style={styles.fetchTextFet}>{contact}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.detailText}>Role:</Text>
            <Text style={styles.fetchTextFet}>{role}</Text>
          </View>
        </View>

        {/* Matches Played Section */}
        <Animated.View style={[styles.card, { transform: [{ translateY: matchesAnim }] }]}>
          <Text style={styles.sectionTitle}>Matches Played</Text>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Matches Played</Text>
            <Text style={styles.cardTextFet}>{matches}</Text>
          </View>
        </Animated.View>

        {/* Batting Section */}
        <Animated.View style={[styles.card, { transform: [{ translateY: battingAnim }] }]}>
          <Text style={styles.sectionTitle}>Batting</Text>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Runs</Text>
            <Text style={styles.cardTextFet}>{battingData.runs}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Balls</Text>
            <Text style={styles.cardTextFet}>{battingData.balls}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Strike Rate</Text>
            <Text style={styles.cardTextFet}>
              {battingData.strikeRate && !isNaN(battingData.strikeRate)
                ? battingData.strikeRate.toFixed(2)
                : 'N/A'}
            </Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Average</Text>
            <Text style={styles.cardTextFet}>
              {battingData.average && !isNaN(battingData.average)
                ? battingData.average.toFixed(2)
                : 'N/A'}
            </Text>
          </View>
        </Animated.View>

        {/* Bowling Section */}
        <Animated.View style={[styles.card, { transform: [{ translateY: bowlingAnim }] }]}>
          <Text style={styles.sectionTitle}>Bowling</Text>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Runs Conceded</Text>
            <Text style={styles.cardTextFet}>{bowlingData.runs}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Balls Bowled</Text>
            <Text style={styles.cardTextFet}>{bowlingData.balls}</Text>
          </View>
          <View style={styles.tableRow}>
            <Text style={styles.cardText}>Wickets</Text>
            <Text style={styles.cardTextFet}>{bowlingData.wickets}</Text>
          </View>
        </Animated.View>
      </ScrollView>

      {/* Modal to select sticker */}
      <Modal
        visible={isModalVisible}
        animationType="fade"
        transparent={true}
        onRequestClose={() => setIsModalVisible(false)}
      >
        <View style={styles.modalContainer}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Select Profile Picture</Text>
            <ScrollView contentContainerStyle={styles.stickerGrid}>
              {imageDictionary.map((image, index) => (
                <TouchableOpacity key={index} onPress={() => handleStickerSelect(image)}>
                  <Image source={image} style={styles.stickerImage} />
                </TouchableOpacity>
              ))}
            </ScrollView>

            {/* Button to pick from gallery */}
            <TouchableOpacity onPress={pickImage} style={styles.customImageButton}>
              <Text style={styles.customImageButtonText}>Choose from Library</Text>
            </TouchableOpacity>

            <TouchableOpacity onPress={() => setIsModalVisible(false)} style={styles.closeButton}>
              <Text style={styles.closeButtonText}>Close</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </Animated.View>
  );
};
